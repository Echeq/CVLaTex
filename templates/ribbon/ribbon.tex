% Template palette: deep blue ribbon, gray support text and gold trim.
\definecolor{accent}{RGB}{31,78,121}
\definecolor{gris}{RGB}{120,130,145}
\definecolor{acento2}{RGB}{191,144,0}

% ---------------------------------------------------------------------------
% Slanted ribbon across the top margin
% ---------------------------------------------------------------------------
% The ribbon is painted by the shipout background hook, so it always stays
% behind the header generated by Python. It is limited to the first page and
% to the top margin: the right end stops just above the text block, therefore
% it never covers the title, the name, the photo or the body.
\newlength{\cvRibbonTop}
\newlength{\cvRibbonBase}
\newlength{\cvRibbonThick}
\newlength{\cvRibbonGold}
\AtBeginDocument{%
  % 1in + \topmargin + heads = distance from the paper edge to the text block.
  \setlength{\cvRibbonTop}{\dimexpr 1in+\topmargin+\headheight+\headsep\relax}%
  % The 1.6 degree slope drops the right end by ~0.028 of the paper width.
  \setlength{\cvRibbonBase}{\dimexpr\cvRibbonTop-0.028\paperwidth-0.16cm\relax}%
  \ifdim\cvRibbonBase<0.45cm \setlength{\cvRibbonBase}{0.45cm}\fi
  \setlength{\cvRibbonThick}{\dimexpr\cvRibbonBase-0.20cm\relax}%
  \ifdim\cvRibbonThick>1.10cm \setlength{\cvRibbonThick}{1.10cm}\fi
  \ifdim\cvRibbonThick<0.30cm \setlength{\cvRibbonThick}{0.30cm}\fi
  \setlength{\cvRibbonGold}{\dimexpr\cvRibbonBase+0.14cm\relax}%
}
\AddToHook{shipout/background}{%
  \ifnum\value{page}=1\relax
    \put(-1cm,-\cvRibbonBase){%
      \rotatebox{-1.6}{\color{accent}\rule{\dimexpr\paperwidth+2cm\relax}{\cvRibbonThick}}}%
    \put(-1cm,-\cvRibbonGold){%
      \rotatebox{-1.6}{\color{acento2}\rule{\dimexpr\paperwidth+2cm\relax}{0.10cm}}}%
  \fi
}

% ---------------------------------------------------------------------------
% Section heading: solid blue bar, white title and a gold rule underneath
% ---------------------------------------------------------------------------
% The bar is a full-width colorbox sized by its own content, so a long title
% wraps inside the bar instead of running into the margin.
\newcommand{\cvsection}[1]{%
  \par\vspace{0.9em}%
  \noindent%
  \begingroup%
  \setlength{\fboxsep}{0pt}%
  \setlength{\fboxrule}{0pt}%
  \colorbox{accent}{%
    \begin{minipage}[b]{\linewidth}%
      \leftskip=0.5em \rightskip=0.5em plus 1fil%
      \vspace*{0.17em}%
      {\color{white}\bfseries\fontsize{13}{15}\selectfont
        \textcolor{acento2}{\rule[-0.06em]{0.14em}{0.92em}}%
        \hspace{0.38em}#1\par}%
      \vspace*{0.17em}%
    \end{minipage}%
  }%
  \endgroup%
  \par\vspace{0.06em}%
  {\color{acento2}\hrule height 1.2pt}%
  \par\vspace{0.42em}%
}

% ---------------------------------------------------------------------------
% Free-block separators, coherent with the blue bar and gold trim
% ---------------------------------------------------------------------------
\newcommand{\cvblockline}{%
  \par\vspace{0.5em}%
  {\color{accent}\hrule height 0.7pt}%
  \par\vspace{0.1em}%
  {\color{acento2}\hrule height 0.4pt}%
  \par\vspace{0.45em}%
}
\newcommand{\cvblockband}{%
  \par\vspace{0.4em}%
  {\color{accent}\hrule height 0.3em}%
  \par\vspace{0.09em}%
  {\color{acento2}\hrule height 0.9pt}%
  \par\vspace{0.4em}%
}
\newcommand{\cvblockspace}{%
  \par\vspace{2em}%
}

% ---------------------------------------------------------------------------
% Three-column entry line: date, title/role and organization/degree
% ---------------------------------------------------------------------------
\newcommand{\cvcnline}[3]{%
  \par\noindent
  % Fixed columns: the date, title and organization keep their vertical axes
  % on every entry. Long Latin/CJK values wrap inside their own column instead
  % of drifting and pushing the line into the right margin. Widths sum to 0.96
  % so the two \hfill gaps keep the middle column centred on the text block.
  \begin{minipage}[t]{0.24\linewidth}%
    \raggedright\textcolor{accent}{\bfseries #1}%
  \end{minipage}%
  \hfill
  \begin{minipage}[t]{0.46\linewidth}%
    \centering{\bfseries #2}%
  \end{minipage}%
  \hfill
  \begin{minipage}[t]{0.26\linewidth}%
    \raggedleft\textcolor{gris}{\small\bfseries #3}%
  \end{minipage}%
  \par\vspace{0.12em}%
}
